users

  • id uuid PK
  • role: khách/CTV/spa/admin
  • phone, email, kyc_status
  • referrer_id FK

stores

  • id PK · owner_id FK
  • slug, theme_json, plan
  • payment_methods[]
  • seo_json

products

  • id PK · category_id
  • name, desc, ingredients
  • price_base, price_min/max
  • media[], seo_json

orders

  • id PK · store_id FK
  • customer_id, items_json
  • payment, shipping, total

landing_pages

  • id PK · store_id, product_id
  • template, content_json
  • pixel_ids, moderation

cms_blocks

  • id PK · page, type
  • content_json, sort_order
  • is_visible, schedule

ai_config

  • id PK · model, api_key
  • system_prompt
  • knowledge_sources[]
  • banned_words[]

ai_conversations

  • id PK · user_id
  • messages_json
  • satisfaction_score
  • led_to_purchase
